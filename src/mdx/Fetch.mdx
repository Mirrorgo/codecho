# fetch API

自己常用的封装 fetch 请求的代码, 处理 get 和 post

- get 的数据都在 params 里面
- post 的数据都在 body 里面

```ts collapse
type ApiResponse<T> = {
  success: boolean;
  code: number;
  data: T;
  msg: string;
};

// get
export const getData = async <T>(
  url: string,
  params: Record<string, string | number> = {}
): Promise<ApiResponse<T>> => {
  const queryString = new URLSearchParams(
    params as Record<string, string>
  ).toString();
  const response = await fetch(`${url}?${queryString}`, {
    credentials: "include",
  });

  if (!response.ok) {
    throw new Error(`Error: ${response.status}`);
  }

  const data: ApiResponse<T> = await response.json();
  return data;
};

// post
export const postData = async <T>(
  url: string,
  body: Record<string, any>
): Promise<ApiResponse<T>> => {
  const response = await fetch(url, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
    },
    credentials: "include",
    body: JSON.stringify(body),
  });

  if (!response.ok) {
    throw new Error(`Error: ${response.status}`);
  }

  const data: ApiResponse<T> = await response.json();
  return data;
};
```

用法

```tsx collapse
type TodoItem = {
  id: number;
  task: string;
  completed: boolean;
};
const [todos, setTodos] = useState<TodoItem[]>([]);
const [loading, setLoading] = useState<boolean>(true);

const fetchTodos = useCallback(async () => {
  try {
    setLoading(true);
    const res = await getData<FundingApplication[]>("/api/todos/list", {
      id: id,
    });
    setTodos(res.data || []);
  } catch (error) {
    console.error("Error fetching todos:", error);
  } finally {
    setLoading(false);
  }
}, [id]);

useEffect(() => {
  fetchTodos();
}, [fetchTodos]);

const handleCreate = async (formData: TodoItem) => {
  try {
    const res = await postData(`/api/todos/add`, formData);
    if (res.success) {
      fetchTodos();
    } else {
      console.error("Create request failed:", res.msg);
    }
  } catch (error) {
    console.error("Error making create request:", error);
  }
};
```

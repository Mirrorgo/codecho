# 受控组件和非受控组件

## 定义方式 1: 是否由 React 组件管理数据

- 受控组件：由 React 组件管理数据
- 非受控组件：由 DOM 元素自身管理数据

> In a controlled component, form data is handled by a React component.  
> The alternative is uncontrolled components, where form data is handled by the DOM itself.  
> 参考：[React 旧版文档](https://legacy.reactjs.org/docs/uncontrolled-components.html)  
> **这种方式不严谨, 更好的方式应该参考定义方式 2**

### 受控组件 (Controlled Component)

<ControlledCounter />

```tsx collapse
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { ChangeEvent, FC, useState } from "react";

const ControlledCounter: FC = () => {
  const [count, setCount] = useState<number>(0);
  const [inputValue, setInputValue] = useState<string>("");

  const handleChange = (e: ChangeEvent<HTMLInputElement>) => {
    setInputValue(e.target.value);
  };

  const handleIncrement = () => {
    const increment = parseInt(inputValue, 10);
    if (!isNaN(increment)) {
      setCount((prevCount) => prevCount + increment);
      setInputValue("");
    }
  };

  return (
    <div className="flex gap-2 items-center">
      <Input
        type="number"
        value={inputValue}
        onChange={handleChange}
        placeholder="Enter number"
        className="w-30"
      />
      <Button onClick={handleIncrement}>Add</Button>
      <div>Count: {count}</div>
    </div>
  );
};

export default ControlledCounter;
```

### 非受控组件 (Uncontrolled Component)

<UncontrolledCounter />

```tsx collapse
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { FC, useRef, useState } from "react";

const UncontrolledCounter: FC = () => {
  const inputRef = useRef<HTMLInputElement>(null);
  const [count, setCount] = useState<number>(0);

  const handleIncrement = () => {
    if (inputRef.current) {
      const increment = parseInt(inputRef.current.value, 10);
      if (!isNaN(increment)) {
        setCount((prevCount) => prevCount + increment);
        inputRef.current.value = ""; // Clear the input field
      }
    }
  };

  return (
    <div className="flex gap-2 items-center">
      <Input
        type="number"
        ref={inputRef}
        placeholder="Enter number"
        className="w-30"
      />
      <Button onClick={handleIncrement}>Add</Button>
      <div>Count: {count}</div>
    </div>
  );
};

export default UncontrolledCounter;
```

## 定义方式 2: 组件状态是否由外部传入 props 完全控制

通常情况下, 组件可能支持两种模式, 受控模式和非受控模式

- 受控模式会支持类似外部传入`value`和`onChange`的 props 来实现
- 非受控模式会支持类似`defaultValue`的 props 来实现, 有可能还会暴露 ref

> 参考: [组件的受控与非受控](https://zhuanlan.zhihu.com/p/536322574)

<InputShow />

> 建议打开 react 开发者工具查看内部状态的变化

### 受控组件 (Controlled Component)

> 支持受控模式

```tsx collapse
import { Input } from "@/components/ui/input";
import { ChangeEvent } from "react";

type ControlledInputProps = {
  value: string;
  onChange: (e: ChangeEvent<HTMLInputElement>) => void;
};
function ControlledInput({ value, onChange }: ControlledInputProps) {
  return <Input value={value} onChange={onChange} className="w-30" />;
}

export default ControlledInput;
```

### 非受控组件 (Uncontrolled Component)

> 支持非受控模式

```tsx collapse
import { Input } from "@/components/ui/input";

type UncontrolledInputProps = {
  defaultValue: string;
};
function UncontrolledInput({ defaultValue }: UncontrolledInputProps) {
  return <Input defaultValue={defaultValue} className="w-30" />;
}

export default UncontrolledInput;
```

### 同时支持受控模式和非受控模式的组件

> 通过**usepropsvalue**来实现  
> 来源: [ant-design-mobile](https://github.com/ant-design/ant-design-mobile/blob/fae45549bcadb2b3c7f1dea27462543230e3b795/src/utils/use-props-value.ts)

```tsx collapse
import { SetStateAction, useRef } from "react";
import { useMemoizedFn, useUpdate } from "ahooks";

type Options<T> = {
  value?: T;
  defaultValue: T;
  onChange?: (v: T) => void;
};

export function usePropsValue<T>(options: Options<T>) {
  const { value, defaultValue, onChange } = options;

  const update = useUpdate();

  const stateRef = useRef<T>(value !== undefined ? value : defaultValue);
  if (value !== undefined) {
    stateRef.current = value;
  }

  const setState = useMemoizedFn(
    (v: SetStateAction<T>, forceTrigger: boolean = false) => {
      // `forceTrigger` means trigger `onChange` even if `v` is the same as `stateRef.current`
      const nextValue =
        typeof v === "function"
          ? (v as (prevState: T) => T)(stateRef.current)
          : v;
      if (!forceTrigger && nextValue === stateRef.current) return;
      stateRef.current = nextValue;
      update();
      onChange?.(nextValue);
    }
  );
  return [stateRef.current, setState] as const;
}
```

用法

```tsx collapse
import React from "react";
import { usePropsValue } from "./usePropsValue";

// 一个简单的输入框组件
function CustomInput({ value, defaultValue = "", onChange }) {
  // 使用 usePropsValue 处理受控和非受控状态
  const [inputValue, setInputValue] = usePropsValue({
    value,
    defaultValue,
    onChange,
  });

  // 处理输入变化
  const handleChange = (e) => {
    setInputValue(e.target.value);
  };

  return <input value={inputValue} onChange={handleChange} />;
}

// 使用示例：
function App() {
  // 非受控模式 - 只提供 defaultValue
  return (
    <div>
      <h3>非受控模式</h3>
      <CustomInput defaultValue="默认文本" />

      <h3>受控模式</h3>
      <ControlledExample />
    </div>
  );
}

// 受控模式示例
function ControlledExample() {
  const [value, setValue] = React.useState("受控文本");

  const handleChange = (newValue) => {
    console.log("值已更新:", newValue);
    setValue(newValue);
  };

  return (
    <>
      <CustomInput value={value} onChange={handleChange} />
      <button onClick={() => setValue("重置文本")}>重置</button>
    </>
  );
}
```

# react api

## `<Profiler>`

用于测量 React 应用中渲染的开销, 可以用于分析组件的渲染性能

```tsx collapse
import React, { Profiler } from "react";

function MyApp() {
  return (
    <Profiler id="导航" onRender={onRenderCallback}>
      <Navigation />
    </Profiler>
  );
}

function onRenderCallback(
  id,
  phase,
  actualDuration,
  baseDuration,
  startTime,
  commitTime,
  interactions
) {
  console.log(`${id} 渲染时间: ${actualDuration}毫秒`);
}
```

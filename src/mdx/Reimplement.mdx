### debounce

```js
function debounce(func, delay) {
  let timeoutId;

  return function (...args) {
    // 清除之前的定时器
    clearTimeout(timeoutId);

    // 设置新的定时器
    timeoutId = setTimeout(() => {
      func.apply(this, args);
    }, delay);
  };
}
```

用法: (打开控制台查看 console 输出)

### throttle

```js
function throttle(func, limit) {
  let lastFunc;
  let lastRan;
  return function (...args) {
    if (!lastRan) {
      func.apply(this, args);
      lastRan = Date.now();
    } else {
      clearTimeout(lastFunc);
      lastFunc = setTimeout(() => {
        if (Date.now() - lastRan >= limit) {
          func.apply(this, args);
          lastRan = Date.now();
        }
      }, limit - (Date.now() - lastRan));
    }
  };
}
```

### flat

```js
function flat(arr, depth = 1) {
  return arr.reduce((acc, cur) => {
    if (Array.isArray(cur) && depth > 0) {
      acc.push(...flat(cur, depth - 1));
    } else {
      acc.push(cur);
    }
    return acc;
  }, []);
}
```

// 递归实现（简单版，不处理循环引用等复杂情况）
function deepCopy(obj) {
  if (typeof obj !== "object" || obj === null) {
    return obj;
  }
  let newObj = Array.isArray(obj) ? [] : {};
  for (let key in obj) {
    if (Object.prototype.hasOwnProperty.call(obj, key)) {
      newObj[key] = deepCopy(obj[key]);
    }
  }
  return newObj;
}

### 日期格式化

实现 `format(new Date(), 'YYYY-MM-DD')` 函数

```js
function format(date, formatString) {
  const year = date.getFullYear(); // 获取年份
  const month = String(date.getMonth() + 1).padStart(2, "0"); // 获取月份并补零
  const day = String(date.getDate()).padStart(2, "0"); // 获取日期并补零

  // 替换格式字符串中的占位符
  return formatString
    .replace("YYYY", year)
    .replace("MM", month)
    .replace("DD", day);
}

// 使用示例
const formattedDate = format(new Date(), "YYYY-MM-DD");
console.log(formattedDate); // 输出: 2024-10-16
```

### 模板引擎

```js
/**
 * 简单的字符串模板引擎
 * 只支持基本变量替换功能
 * 使用方式: "Hello, {name}".use({ name: "World" })
 */
String.prototype.use = function (data) {
  if (!data) return this;
  return this.replace(/{([^{}]+)}/g, function (match, key) {
    return data[key] !== undefined ? data[key] : match;
  });
};
```

### 手写 Promise.all

原本效果

```js
const promise1 = Promise.resolve(3);
const promise2 = 42;
const promise3 = new Promise((resolve, reject) => {
  setTimeout(resolve, 100, "foo");
});

Promise.all([promise1, promise2, promise3]).then((values) => {
  console.log(values);
});
```

手写实现

```js collapse
function promiseAll(promises) {
  if (promises.length === 0) {
    return Promise.resolve([]);
  }

  return new Promise((resolve, reject) => {
    const results = new Array(promises.length);
    let completedCount = 0;

    promises.forEach((promise, index) => {
      Promise.resolve(promise)
        .then((value) => {
          results[index] = value;
          if (++completedCount === promises.length) {
            resolve(results);
          }
        })
        .catch(reject);
    });
  });
}
```

用法

```js collapse
// 测试用例

const promise2 = new Promise((resolve) => setTimeout(() => resolve(2), 100));
const promise3 = new Promise((resolve) => setTimeout(() => resolve(3), 50));

  .then((values) => {
    console.log("所有Promise已完成:", values); // [1, 2, 3]
  })
  .catch((error) => {
    console.error("有Promise失败:", error);
  });

// 失败的情况

    console.log("所有Promise已完成:", values);
  })
  .catch((error) => {
    console.error("有Promise失败:", error); // 'Promise失败了'
  });
```

### 手写 Promise.any

原本效果

```js
const promise1 = Promise.reject(0);
const promise2 = new Promise((resolve) => setTimeout(resolve, 100, "quick"));
const promise3 = new Promise((resolve) => setTimeout(resolve, 500, "slow"));

const promises = [promise1, promise2, promise3];

Promise.any(promises).then((value) => console.log(value));
// Expected output: "quick"
```

手写实现

```js collapse
function promiseAny(promises) {
  return new Promise((resolve, reject) => {
    if (promises.length === 0) {
      reject(new AggregateError([], "All promises were rejected"));
      return;
    }

    let rejectedCount = 0;
    const errors = [];

    promises.forEach((promise) => {
      Promise.resolve(promise)
        .then((value) => {
          resolve(value);
        })
        .catch((error) => {
          errors.push(error);
          rejectedCount++;

          if (rejectedCount === promises.length) {
            reject(new AggregateError(errors, "All promises were rejected"));
          }
        });
    });
  });
}
```

用法

```js collapse
// 简单示例
const promise1 = Promise.reject("错误1");
const promise2 = new Promise((resolve) =>
  setTimeout(() => resolve("成功"), 100)
);
const promise3 = Promise.reject("错误2");

promiseAny([promise1, promise2, promise3])
  .then((value) => console.log(value)) // 输出: "成功"
  .catch((error) => console.log(error));

// 全部拒绝的例子
promiseAny([Promise.reject("错误1"), Promise.reject("错误2")])
  .then((value) => console.log(value))
  .catch((error) => console.log("全部拒绝:", error.message, error.errors)); // 输出错误信息
```

### 手写 Promise.race

原本效果

```js
const promise1 = new Promise((resolve, reject) => {
  setTimeout(resolve, 500, "one");
});

const promise2 = new Promise((resolve, reject) => {
  setTimeout(resolve, 100, "two");
});

Promise.race([promise1, promise2]).then((value) => {
  console.log(value);
  // Both resolve, but promise2 is faster
});
// Expected output: "two"
```

手写实现

```js collapse
function promiseRace(promises) {
  return new Promise((resolve, reject) => {
    if (promises.length === 0) {
      return; // pend forever, same as native Promise.race
    }

    promises.forEach((promise) => {
      Promise.resolve(promise)
        .then((value) => {
          resolve(value);
        })
        .catch((error) => {
          reject(error);
        });
    });
  });
}
```

用法

```js collapse
// 示例：带有成功和失败的 Promises
const promise1 = new Promise((resolve) =>
  setTimeout(() => resolve("第一个完成"), 200)
);
const promise2 = new Promise((resolve) =>
  setTimeout(() => resolve("第二个完成"), 100)
);
const promise3 = new Promise((resolve, reject) =>
  setTimeout(() => reject("第三个失败"), 50)
);

// 测试成功的场景
promiseRace([promise1, promise2])
  .then((value) => console.log("成功:", value)) // 输出: "成功: 第二个完成"
  .catch((error) => console.log("失败:", error));

// 测试失败的场景
promiseRace([promise1, promise3])
  .then((value) => console.log("成功:", value))
  .catch((error) => console.log("失败:", error)); // 输出: "失败: 第三个失败"

// 测试空数组
const emptyRace = promiseRace([]);
console.log("空数组的 race 将永远处于 pending 状态:", emptyRace);
```
